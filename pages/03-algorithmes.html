Les 10 algorithmes du TSP

📌 Note : Chaque algorithme est implémenté en Python. Les codes sont prêts à être copiés et exécutés.
1. Force Brute (Brute Force)
Exact O(n!)
Principe : Énumérer toutes les permutations possibles des villes, calculer la longueur de chaque tournée, et garder la meilleure. C'est la méthode la plus simple mais aussi la plus lente. 
def tsp_bruteforce(dist, n):
    from itertools import permutations
    villes = list(range(n))
    best_cost = float('inf')
    best_path = []
    for perm in permutations(villes):
        cost = sum(dist[perm[i]][perm[i+1]] for i in range(n-1))
        cost += dist[perm[-1]][perm[0]]
        if cost < best_cost:
            best_cost = cost
            best_path = list(perm)
    return best_path, best_cost
✅ Avantage : Garantie d'optimalité.
❌ Inconvénient : Complexité factorielle O(n!) — inutilisable pour n > 12.
2. Held-Karp (Programmation Dynamique)
Exact O(n²·2ⁿ)
Principe : Utiliser la programmation dynamique avec un masque de bits. Pour chaque sous-ensemble S de villes et chaque ville j ∈ S, on stocke le coût minimal d'un chemin partant de la ville 0, visitant toutes les villes de S et se terminant à j. 
def tsp_held_karp_with_path(dist, n):
    dp = [[float('inf')] * n for _ in range(1 << n)]
    parent = [[-1] * n for _ in range(1 << n)]
    dp[1][0] = 0
    for mask in range(1 << n):
        for i in range(n):
            if dp[mask][i] == float('inf'): continue
            for j in range(n):
                if not (mask >> j) & 1:
                    nouveau_mask = mask | (1 << j)
                    nouveau_cout = dp[mask][i] + dist[i][j]
                    if nouveau_cout < dp[nouveau_mask][j]:
                        dp[nouveau_mask][j] = nouveau_cout
                        parent[nouveau_mask][j] = i
    mask_complet = (1 << n) - 1
    best_cost = float('inf')
    last_city = -1
    for i in range(n):
        total = dp[mask_complet][i] + dist[i][0]
        if total < best_cost:
            best_cost = total
            last_city = i
    path = []
    mask = mask_complet
    city = last_city
    while city != -1:
        path.append(city)
        prev_city = parent[mask][city]
        mask = mask ^ (1 << city)
        city = prev_city
    path.reverse()
    return path, best_cost
✅ Avantage : Optimal et bien meilleur que O(n!) (20 villes : 4×10⁸ vs 2×10¹⁸).
❌ Inconvénient : Mémoire O(n·2ⁿ) — limité à n≈25-30.
3. Plus Proche Voisin (Nearest Neighbor)
Heuristique O(n²)
Principe : Partir d'une ville, puis à chaque étape, aller vers la ville non visitée la plus proche. Très simple et rapide. 
def tsp_nearest_neighbor(dist, start=0):
    n = len(dist)
    visited = [False] * n
    path = [start]
    visited[start] = True
    current = start
    for _ in range(n-1):
        nxt = min((i for i in range(n) if not visited[i]), key=lambda i: dist[current][i])
        path.append(nxt)
        visited[nxt] = True
        current = nxt
    cost = sum(dist[path[i]][path[i+1]] for i in range(n-1)) + dist[path[-1]][start]
    return path, cost
✅ Avantage : Très rapide O(n²).
❌ Inconvénient : Solution souvent 20-30% au-dessus de l'optimum.
4. Insertion du Plus Loin (Farthest Insertion)
Heuristique O(n²)
Principe : Commencer par 2 villes, puis insérer à chaque étape la ville la plus éloignée du cycle actuel, à la position qui minimise l'augmentation du coût. 
def tsp_farthest_insertion(dist):
    n = len(dist)
    cycle = [0]
    remaining = set(range(1, n))
    farthest = max(remaining, key=lambda x: dist[0][x])
    cycle.append(farthest)
    remaining.remove(farthest)
    while remaining:
        best_dist = -1
        best_city = None
        best_pos = None
        for city in remaining:
            min_insert_cost = float('inf')
            insert_pos = 0
            for i in range(len(cycle)):
                a = cycle[i]
                b = cycle[(i+1) % len(cycle)]
                cost = dist[a][city] + dist[city][b] - dist[a][b]
                if cost < min_insert_cost:
                    min_insert_cost = cost
                    insert_pos = i+1
            if min_insert_cost > best_dist:
                best_dist = min_insert_cost
                best_city = city
                best_pos = insert_pos
        cycle.insert(best_pos, best_city)
        remaining.remove(best_city)
    cost = sum(dist[cycle[i]][cycle[(i+1) % n]] for i in range(n))
    return cycle, cost
5. 2-OPT (Amélioration locale)
Heuristique O(n³)
Principe : Prendre un chemin existant, inverser un segment entre deux positions. Si l'inversion réduit la distance, on garde le nouveau chemin. Répéter jusqu'à stabilisation. 
def two_opt(route, dist):
    improved = True
    best = route
    n = len(route)
    def path_cost(r):
        return sum(dist[r[i]][r[(i+1)%n]] for i in range(n))
    best_cost = path_cost(best)
    while improved:
        improved = False
        for i in range(1, n-1):
            for j in range(i+1, n):
                if j - i == 1: continue
                new_route = best[:i] + best[i:j+1][::-1] + best[j+1:]
                new_cost = path_cost(new_route)
                if new_cost < best_cost:
                    best = new_route
                    best_cost = new_cost
                    improved = True
                    break
            if improved: break
        if not improved: break
    return best, best_cost
6. Christofides (Approximation garantie)
Approché O(n³)
Principe : Construire un arbre couvrant minimal (MST), trouver les sommets de degré impair, faire un couplage minimal entre eux, ajouter ces arêtes, construire un circuit eulérien, puis le court-circuiter. 
def tsp_christofides(dist):
    n = len(dist)
    # MST par Prim
    mst = [[] for _ in range(n)]
    in_mst = [False] * n
    min_edge = [float('inf')] * n
    parent = [-1] * n
    min_edge[0] = 0
    for _ in range(n):
        u = min((i for i in range(n) if not in_mst[i]), key=lambda x: min_edge[x])
        in_mst[u] = True
        if parent[u] != -1:
            mst[parent[u]].append(u)
            mst[u].append(parent[u])
        for v in range(n):
            if not in_mst[v] and dist[u][v] < min_edge[v]:
                min_edge[v] = dist[u][v]
                parent[v] = u
    # Sommets de degré impair
    degree = [len(mst[i]) for i in range(n)]
    odd = [i for i in range(n) if degree[i] % 2 == 1]
    # Couplage glouton
    matched = [False] * n
    matching = []
    for i in range(len(odd)):
        if matched[odd[i]]: continue
        best_j, best_d = -1, float('inf')
        for j in range(i+1, len(odd)):
            if not matched[odd[j]] and dist[odd[i]][odd[j]] < best_d:
                best_d = dist[odd[i]][odd[j]]
                best_j = odd[j]
        if best_j != -1:
            matching.append((odd[i], best_j))
            matched[odd[i]] = matched[best_j] = True
    for u, v in matching:
        mst[u].append(v); mst[v].append(u)
    # Circuit eulérien
    stack, path = [0], []
    while stack:
        v = stack[-1]
        if mst[v]:
            u = mst[v].pop()
            mst[u].remove(v)
            stack.append(u)
        else:
            path.append(stack.pop())
    # Shortcut
    visited = [False] * n
    tsp_path = []
    for v in path:
        if not visited[v]:
            visited[v] = True
            tsp_path.append(v)
    cost = sum(dist[tsp_path[i]][tsp_path[(i+1)%n]] for i in range(n))
    return tsp_path, cost
✅ Garantie : Coût ≤ 1.5 × optimum (pour TSP métrique).
7. Colonies de Fourmis (Ant Colony Optimization)
Métaheuristique O(iterations × ants × n²)
Principe : Inspiré du comportement des fourmis. Chaque fourmi construit une tournée en choisissant la prochaine ville selon la phéromone déposée et la proximité. On dépose plus de phéromone sur les meilleurs chemins. 
import random
def tsp_aco(dist, n, ants=20, alpha=1, beta=3, rho=0.3, q0=0.9, iterations=200):
    pheromone = [[1.0] * n for _ in range(n)]
    best_path, best_cost = None, float('inf')
    for _ in range(iterations):
        paths, costs = [], []
        for _ in range(ants):
            path, visited = [0], [False]*n
            visited[0] = True
            for _ in range(n-1):
                curr = path[-1]
                candidates = [(nxt, pheromone[curr][nxt], 1.0/dist[curr][nxt]) 
                              for nxt in range(n) if not visited[nxt]]
                if random.random() < q0:
                    nxt = max(candidates, key=lambda x: (x[1]**alpha)*(x[2]**beta))[0]
                else:
                    total = sum((tau**alpha)*(eta**beta) for _, tau, eta in candidates)
                    r = random.random()
                    cum = 0
                    for nxt, tau, eta in candidates:
                        cum += (tau**alpha)*(eta**beta)/total
                        if r <= cum: break
                path.append(nxt); visited[nxt] = True
            cost = sum(dist[path[i]][path[i+1]] for i in range(n-1)) + dist[path[-1]][path[0]]
            paths.append(path); costs.append(cost)
            if cost < best_cost: best_cost, best_path = cost, path.copy()
        for i in range(n):
            for j in range(n): pheromone[i][j] *= (1-rho)
        for path, cost in zip(paths, costs):
            deposit = 1.0/cost
            for i in range(n-1): pheromone[path[i]][path[i+1]] += deposit
            pheromone[path[-1]][path[0]] += deposit
    return best_path, best_cost
8. Algorithme Génétique (Genetic Algorithm)
Métaheuristique O(generations × pop_size × n)
Principe : Population de solutions, sélection des meilleurs, croisement (crossover), mutation, et remplacement. Évolution inspirée de la biologie. 
import random
def tsp_genetic(dist, n, pop_size=100, generations=500, mutation_rate=0.1):
    def path_cost(route):
        return sum(dist[route[i]][route[(i+1)%n]] for i in range(n))
    def create(): r = list(range(n)); random.shuffle(r); return r
    def crossover(p1, p2):
        start, end = sorted(random.sample(range(n), 2))
        child = [-1]*n
        child[start:end] = p1[start:end]
        pos = end
        for g in p2:
            if g not in child:
                if pos >= n: pos = 0
                child[pos] = g; pos += 1
        return child
    def mutate(r):
        if random.random() < mutation_rate:
            i, j = random.sample(range(n), 2)
            r[i], r[j] = r[j], r[i]
        return r
    pop = [create() for _ in range(pop_size)]
    for _ in range(generations):
        pop.sort(key=path_cost)
        new_pop = pop[:pop_size//2]
        while len(new_pop) < pop_size:
            p1 = random.choice(pop[:pop_size//4])
            p2 = random.choice(pop[:pop_size//4])
            child = mutate(crossover(p1, p2))
            new_pop.append(child)
        pop = new_pop
    best = min(pop, key=path_cost)
    return best, path_cost(best)
9. Lin-Kernighan (Local Search)
Métaheuristique O(n²·iterations)
Principe : Version améliorée du 2-opt qui échange plusieurs paires d'arêtes à chaque étape pour une meilleure optimisation locale. 
def tsp_lin_kernighan(dist, route):
    n = len(route)
    def path_cost(r):
        return sum(dist[r[i]][r[(i+1)%n]] for i in range(n))
    improved = True
    best_route = route.copy()
    best_cost = path_cost(best_route)
    while improved:
        improved = False
        for i in range(n):
            for j in range(i+2, n):
                if j == n-1 and i == 0: continue
                new_route = best_route.copy()
                new_route[i+1:j+1] = reversed(best_route[i+1:j+1])
                new_cost = path_cost(new_route)
                if new_cost < best_cost:
                    best_route = new_route
                    best_cost = new_cost
                    improved = True
                    break
            if improved: break
        if not improved: break
    return best_route, best_cost
10. Branch & Bound (Séparation et Évaluation)
Exact Exponentiel (mais élague)
Principe : Exploration intelligente de l'arbre de recherche. À chaque nœud, on calcule une borne inférieure. Si cette borne est ≥ meilleure solution connue, on élague la branche. 
def tsp_branch_bound_simple(dist, n):
    best = [float('inf')]
    best_route = [None]
    def lower_bound(path, cost):
        visited = [False]*n
        for v in path: visited[v] = True
        remaining = [i for i in range(n) if not visited[i]]
        if not remaining:
            return cost + dist[path[-1]][path[0]]
        lb = cost + min(dist[path[-1]][i] for i in remaining) if remaining else 0
        for v in remaining:
            lb += min(dist[v][i] for i in range(n) if i != v)
        return lb
    def search(path, cost):
        if len(path) == n:
            total = cost + dist[path[-1]][path[0]]
            if total < best[0]:
                best[0] = total
                best_route[0] = path.copy()
            return
        last = path[-1]
        for nxt in range(n):
            if nxt not in path:
                new_cost = cost + dist[last][nxt]
                if new_cost + lower_bound(path+[nxt], new_cost) < best[0]:
                    search(path+[nxt], new_cost)
    search([0], 0)
    return best_route[0], best[0]
✅ Avantage : Peut résoudre des instances jusqu'à n≈35-40.
❌ Inconvénient : Pire cas exponentiel.
